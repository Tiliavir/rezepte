{{- /* Security policy as because GitHub Pages cannot send custom response headers. Must stay directly after : a meta CSP only applies to content that follows it. __CSP_SCRIPT_HASHES__ is replaced per page by csp-hashes.mjs during "npm run build". Everything (styles, fonts, scripts, search index) is served from this site; no third-party requests. Not possible via : frame-ancestors/X-Frame-Options, HSTS, X-Content-Type-Options, Permissions-Policy. */ -}} {{- $u := urls.Parse site.BaseURL -}} {{- $self := printf "'self' %s://%s" $u.Scheme $u.Host -}} {{- $csp := slice (printf "default-src %s" $self) (printf "script-src %s __CSP_SCRIPT_HASHES__" $self) (printf "style-src %s" $self) (printf "img-src %s data:" $self) "object-src 'none'" (printf "base-uri %s" $self) (printf "form-action %s" $self) -}} {{- /* hugo server: no hashing step runs and it is plain http, so allow the inline theme script and skip the upgrade */ -}} {{- if hugo.IsServer }}{{ $csp = $csp | append "script-src-elem 'self' 'unsafe-inline'" }}{{ else }}{{ $csp = $csp | append "upgrade-insecure-requests" }}{{ end -}} {{ printf "" (delimit $csp "; ") | safeHTML }}
